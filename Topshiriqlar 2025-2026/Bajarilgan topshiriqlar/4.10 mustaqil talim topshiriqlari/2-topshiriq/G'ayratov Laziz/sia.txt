sonlar = []

son = float(input("Son kiriting (0 kiritilsa tugaydi): "))
while son != 0:
    sonlar.append(son)
    son = float(input("Son kiriting (0 kiritilsa tugaydi): "))

if sonlar:
    print(f"Kiritilgan sonlar soni: {len(sonlar)}")
    print(f"O'rtacha arifmetik: {sum(sonlar)/len(sonlar)}")
else:
    print("Hech qanday son kiritilmadi.")
